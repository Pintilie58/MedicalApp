// ============================================================================
//  End-to-end probe for the split pipeline (chunked extraction A, parallel
//  explanation batches B, narrative C, completeness sweep A2) against a LOCAL
//  fake Gemini, plus unit checks for the deterministic out-of-range completer.
// ============================================================================

int fails = 0;
void Check(string label, bool ok, string detail = "")
{
    Console.WriteLine((ok ? "PASS  " : "FAIL  ") + label + (detail.Length > 0 ? "  ->  " + detail : ""));
    if (!ok) fails++;
}

const int Port = 5601;
const int AnalyteCount = 40;

var recorded = new List<Call>();
var recordLock = new object();
var flags = new Flags();

var listener = new HttpListener();
listener.Prefixes.Add($"http://localhost:{Port}/");
listener.Start();

_ = Task.Run(async () =>
{
    while (listener.IsListening)
    {
        HttpListenerContext ctx;
        try { ctx = await listener.GetContextAsync(); } catch { return; }

        _ = Task.Run(async () =>
        {
            var started = DateTime.UtcNow;
            using var reader = new StreamReader(ctx.Request.InputStream, Encoding.UTF8);
            var body = await reader.ReadToEndAsync();
            var model = ctx.Request.Url!.AbsolutePath.Split('/').Last().Replace(":generateContent", "");

            using var doc = JsonDocument.Parse(body);
            var userText = doc.RootElement.GetProperty("contents")[0]
                              .GetProperty("parts")[0].GetProperty("text").GetString() ?? "";
            var sysText = doc.RootElement.GetProperty("systemInstruction")
                             .GetProperty("parts")[0].GetProperty("text").GetString() ?? "";
            var genCfg = doc.RootElement.GetProperty("generationConfig");
            string? level = null, budget = null;
            if (genCfg.TryGetProperty("thinkingConfig", out var tc))
            {
                if (tc.TryGetProperty("thinkingLevel", out var lv)) level = lv.GetString();
                if (tc.TryGetProperty("thinkingBudget", out var bd)) budget = bd.GetRawText();
            }

            int fragment = 0;
            var fm = System.Text.RegularExpressions.Regex.Match(userText, @"FRAGMENT (\d) OF (\d)");
            if (fm.Success) fragment = int.Parse(fm.Groups[1].Value);

            string stage = userText.Contains("COMPLETENESS SWEEP") ? "SWEEP"
                         : userText.Contains("STAGE 1 OF 3") ? "A"
                         : userText.Contains("STAGE 2 OF 3") ? "B"
                         : userText.Contains("STAGE 3 OF 3") ? "C"
                         : "MONO";

            bool fail = (stage == "A" && fragment == 2 && flags.FailSecondChunk)
                     || (stage == "C" && flags.FailStageC);

            await Task.Delay(200);

            string payload;
            if (fail)
            {
                ctx.Response.StatusCode = 500;
                payload = "{\"error\":{\"message\":\"simulated failure\"}}";
            }
            else
            {
                ctx.Response.StatusCode = 200;
                payload = BuildReply(stage, userText, fragment, flags.SweepFinds);
            }

            var finished = DateTime.UtcNow;
            lock (recordLock) recorded.Add(new Call(stage, model, level, budget, started, finished, userText, sysText, fragment));

            var bytes = Encoding.UTF8.GetBytes(payload);
            ctx.Response.ContentType = "application/json";
            await ctx.Response.OutputStream.WriteAsync(bytes);
            ctx.Response.Close();
        });
    }
});

static string Wrap(string innerJson) =>
    JsonSerializer.Serialize(new
    {
        candidates = new[]
        {
            new { finishReason = "STOP", content = new { parts = new[] { new { text = innerJson } } } }
        },
        usageMetadata = new { promptTokenCount = 1000, candidatesTokenCount = 500, thoughtsTokenCount = 120 }
    });

static object Analyte(int i, bool withExplanation) => new Dictionary<string, object?>
{
    ["parameter"] = $"Analiza {i}",
    ["value"] = (10 + i).ToString(),
    ["unit"] = "mg/dL",
    ["reference_range"] = "0-100",
    ["status"] = i % 5 == 0 ? "high" : "normal",
    ["explanation"] = withExplanation ? $"Explicatie monolitica {i}" : "",
    ["parameter_normalized_en"] = $"Analyte {i} [Mass/volume] in Serum",
    ["panel_header_raw"] = "Biochimie"
};

static string BuildReply(string stage, string userText, int fragment, bool sweepFinds)
{
    if (stage == "A" || stage == "MONO")
    {
        // Fragment 1 covers 0..20 (21 analytes, one of them shared with the
        // second fragment => the overlap duplicate), fragment 2 covers 20..39.
        IEnumerable<int> range = fragment switch
        {
            1 => Enumerable.Range(0, 21),
            2 => Enumerable.Range(20, AnalyteCount - 20),
            _ => Enumerable.Range(0, AnalyteCount)
        };
        var krs = range.Select(i => Analyte(i, stage == "MONO")).ToList();

        var obj = new Dictionary<string, object?>
        {
            ["is_medical_analysis"] = true,
            ["patient_info"] = fragment == 2
                ? new { name = (string?)null, date_taken = (string?)null, laboratory = (string?)null }
                : new { name = "Test Pacient", date_taken = "10/06/2026", laboratory = "Lab Probe" },
            ["summary"] = stage == "MONO" ? "Sumar monolitic" : "",
            ["key_results"] = krs,
            ["abnormal_findings"] = stage == "MONO"
                ? new object[] { new { parameter = "Analiza 0", explanation = "mono", severity = "mild" } }
                : Array.Empty<object>(),
            ["correlations"] = stage == "MONO" ? "Corelatii monolitice" : "",
            ["recommendations"] = stage == "MONO" ? "Recomandari monolitice" : "",
            ["disclaimer"] = stage == "MONO" ? "Disclaimer mono" : "",
            ["_extraction_audit"] = new { expected_count = krs.Count, parameter_names = Array.Empty<string>() }
        };
        return Wrap(JsonSerializer.Serialize(obj));
    }

    if (stage == "SWEEP")
        return sweepFinds
            ? Wrap(JsonSerializer.Serialize(new
            {
                missing = new[]
                {
                    new { parameter = "Analiza recuperata", value = "77", unit = "mg/dL",
                          reference_range = "0-100", status = "normal", explanation = "" }
                }
            }))
            : Wrap("{\"missing\":[]}");

    if (stage == "B")
    {
        var indices = System.Text.RegularExpressions.Regex.Matches(userText, @"^\[(\d+)\] ",
                          System.Text.RegularExpressions.RegexOptions.Multiline)
                      .Select(m => int.Parse(m.Groups[1].Value)).ToList();
        return Wrap(JsonSerializer.Serialize(new
        {
            explanations = indices.Select(i => new { index = i, explanation = $"Explicatie etapa B pentru {i}" })
        }));
    }

    return Wrap(JsonSerializer.Serialize(new
    {
        summary = "Sumar din etapa C",
        abnormal_findings = new[] { new { parameter = "Analiza 0", explanation = "anormal C", severity = "moderate" } },
        correlations = "Corelatii din etapa C",
        recommendations = "Recomandari din etapa C",
        disclaimer = "Disclaimer C",
        risk_factors = new[] { "risc C" },
        doctor_questions = new[] { "intrebare C" }
    }));
}

GeminiMedicalInterpretationService Build(string pipelineMode, int chunks = 2, bool sweep = true)
{
    var settings = new GeminiSettings
    {
        ApiKey = "test-key",
        BaseUrl = $"http://localhost:{Port}/v1beta",
        Model = "gemini-2.5-flash",
        PipelineMode = pipelineMode,
        ExtractorModel = "gemini-2.5-flash",
        ExplainModel = "gemini-2.5-flash",
        NarrativeModel = "gemini-2.5-flash",
        ExtractorThinkingLevel = "low",
        ExplainThinkingLevel = "minimal",
        NarrativeThinkingLevel = "high",
        ExtractorChunks = chunks,
        ExplainBatchSize = 12,
        EnableCompletenessSweep = sweep,
        TimeoutSeconds = 30,
        MaxOutputTokens = 65000,
        ThinkingBudget = -1
    };
    return new GeminiMedicalInterpretationService(
        Options.Create(settings),
        new ProbeHttpClientFactory(),
        LoggerFactory.Create(b => b.SetMinimumLevel(LogLevel.Error))
            .CreateLogger<GeminiMedicalInterpretationService>());
}

// A report text long enough to be worth chunking (>4000 chars, >40 lines).
var text = "LABORATOR TEST\nPacient: Test\nData: 10/06/2026\n" +
           string.Join("\n", Enumerable.Range(0, 12).Select(i => $"Sectiune informativa linia {i} — text de umplere pentru antet.")) + "\n" +
           string.Join("\n", Enumerable.Range(0, AnalyteCount)
               .Select(i => $"Analiza {i}    {10 + i}   mg/dL   0-100    metoda enzimatica, laborator intern, comentariu {i}"));

// ================= 1. SPLIT with CHUNKED extraction =========================
flags.FailSecondChunk = false; flags.FailStageC = false; flags.SweepFinds = true;
recorded.Clear();
var svc = Build("split");

// Live progress hook (what the upload screen polls): the table must be
// published as soon as the extraction stage is done, LONG before the end.
var progressEvents = new List<(string Stage, int Count, DateTime At)>();
var runStart = DateTime.UtcNow;
svc.OnStage = (stage, analytes) => progressEvents.Add((stage, analytes?.Count ?? 0, DateTime.UtcNow));

var (res, inTok, outTok, raw) = await svc.InterpretTextAsync(text, "probe.pdf", "ro");
var runEnd = DateTime.UtcNow;

Check("progress: the table is published once, right after extraction",
    progressEvents.Count(e => e.Stage == "ai_extract_done") == 1,
    string.Join(",", progressEvents.Select(e => e.Stage + "(" + e.Count + ")")));
Check("progress: the published table already holds all 40 extracted analytes",
    progressEvents.First(e => e.Stage == "ai_extract_done").Count == 40,
    progressEvents.First(e => e.Stage == "ai_extract_done").Count.ToString());
Check("progress: published BEFORE the pipeline finished (this is the whole point)",
    progressEvents.First(e => e.Stage == "ai_extract_done").At < runEnd.AddMilliseconds(-150),
    $"published at +{(progressEvents.First(e => e.Stage == "ai_extract_done").At - runStart).TotalMilliseconds:0}ms of {(runEnd - runStart).TotalMilliseconds:0}ms");

// The tracker itself (what the JSON endpoint returns)
var tracker = new InterpretationProgressTracker();
tracker.SetStage("tok1", "ai_extract");
Check("tracker: stage stored", tracker.Get("tok1")!.Stage == "ai_extract");
tracker.SetTable("tok1", new List<KeyResult>
{
    new() { Parameter = "Colesterol", Value = "250", Unit = "mg/dL", ReferenceRange = "0-200", Status = "HIGH" },
    new() { Parameter = "Glucoza", Value = "90", Unit = "mg/dL", ReferenceRange = "70-100", Status = "normal" },
    new() { Parameter = "  ", Value = "1", Status = "low" }
});
var st1 = tracker.Get("tok1")!;
Check("tracker: table published, blank names skipped, status normalized",
    st1.Table!.Count == 2 && st1.Table[0].Status == "high", st1.Table!.Count.ToString());
Check("tracker: out-of-range counted", st1.OutOfRangeCount == 1, st1.OutOfRangeCount.ToString());
Check("tracker: stage advances to the explanation step", st1.Stage == "ai_explain");
tracker.Fail("tok1", "boom");
Check("tracker: failure recorded", tracker.Get("tok1")!.Stage == "error" && tracker.Get("tok1")!.Error == "boom");
Check("tracker: unknown token returns null (endpoint answers 'upload')", tracker.Get("nope") == null);
Check("tracker: null/empty token is a no-op", tracker.Get(null) == null);

List<Call> St(string s) => recorded.Where(c => c.Stage == s).ToList();

Check("chunked A: 2 parallel extraction calls", St("A").Count == 2, "A=" + St("A").Count);
Check("chunked A: the two calls really overlap in time",
    St("A")[0].Started < St("A")[1].Finished && St("A")[1].Started < St("A")[0].Finished);
Check("chunked A: each call received a FRAGMENT note", St("A").All(c => c.Fragment > 0),
    string.Join("/", St("A").Select(c => c.Fragment.ToString())));
Check("chunked A: telemetry records 2 chunks", svc.LastStageTimings.GetValueOrDefault("ai_a_chunks") == 2);
Check("chunked A: tables stitched to 40 unique analytes + 1 recovered = 41",
    res.KeyResults?.Count == 41, "count=" + (res.KeyResults?.Count ?? 0));
Check("chunked A: the overlap duplicate was dropped",
    res.KeyResults!.Count(k => k.Parameter == "Analiza 20") == 1,
    res.KeyResults!.Count(k => k.Parameter == "Analiza 20").ToString());
Check("chunked A: no analyte lost across the cut",
    Enumerable.Range(0, AnalyteCount).All(i => res.KeyResults!.Any(k => k.Parameter == $"Analiza {i}")));
Check("chunked A: patient info recovered from the fragment that had it",
    res.PatientInfo?.Laboratory == "Lab Probe" && res.PatientInfo?.Name == "Test Pacient");
Check("chunked A: audit parameter_names dropped (Fix 4)", res.Audit?.ParameterNames == null);
Check("chunked A: audit expected_count equals the final table (40 stitched + 1 recovered)", res.Audit?.ExpectedCount == 41,
    res.Audit?.ExpectedCount.ToString() ?? "null");

Check("split: sweep called once and recovered 1 analyte",
    St("SWEEP").Count == 1 && svc.LastStageTimings.GetValueOrDefault("ai_s_recovered") == 1);
Check("split: explanations for all 41 analytes",
    res.KeyResults!.All(k => !string.IsNullOrWhiteSpace(k.Explanation)),
    res.KeyResults!.Count(k => string.IsNullOrWhiteSpace(k.Explanation)) + " missing");
Check("split: narrative merged from stage C", res.Summary == "Sumar din etapa C");
Check("split: no monolithic fallback", St("MONO").Count == 0);

// --- Fix 1: stage C on flash with thinking HIGH
var cCall = St("C")[0];
Check("Fix 1: stage C runs on gemini-2.5-flash with thinkingBudget=16384 (high)",
    cCall.Model == "gemini-2.5-flash" && cCall.ThinkingBudget == "16384" && cCall.ThinkingLevel == null,
    $"{cCall.Model} budget={cCall.ThinkingBudget}");

// --- Fix 3: short system prompt for stage B
var aSys = St("A")[0].SystemPrompt;
var bSys = St("B")[0].SystemPrompt;
Check("Fix 3: stage B system prompt is much shorter than stage A's",
    bSys.Length < aSys.Length / 3, $"B={bSys.Length} chars vs A={aSys.Length} chars");
Check("Fix 3: stage B prompt keeps the explanation-depth rules VERBATIM",
    bSys.Contains("EXPLANATION DEPTH POLICY")
    && bSys.Contains("EXACTLY 1-2 SHORT")
    && bSys.Contains("AT LEAST 3 FULL SENTENCES")
    && bSys.Contains("MULTI-THRESHOLD / TIERED-TARGET RULE"));
Check("Fix 3: stage B prompt drops the extraction rules it does not need",
    !bSys.Contains("EXTRACTION COMPLETENESS") && !bSys.Contains("SELF-VERIFICATION FIELD"));
Check("Fix 3: stage C keeps the FULL prompt (single call, quality first)",
    cCall.SystemPrompt.Contains("EXTRACTION COMPLETENESS"));

// ================= 2. A chunk fails -> single full-text extraction ==========
flags.FailSecondChunk = true; flags.SweepFinds = false;
recorded.Clear();
var svc2 = Build("split");
var (res2, _, _, _) = await svc2.InterpretTextAsync(text, "probe.pdf", "ro");
Check("chunk failure: retried as ONE full-text extraction",
    recorded.Count(c => c.Stage == "A" && c.Fragment == 0) == 1,
    "unchunked A=" + recorded.Count(c => c.Stage == "A" && c.Fragment == 0));
Check("chunk failure: telemetry says 1 chunk", svc2.LastStageTimings.GetValueOrDefault("ai_a_chunks") == 1);
Check("chunk failure: complete report anyway, no monolithic fallback",
    res2.KeyResults?.Count == 40 && res2.Summary == "Sumar din etapa C" && recorded.All(c => c.Stage != "MONO"),
    "count=" + (res2.KeyResults?.Count ?? 0));

// ================= 3. Chunking disabled =====================================
flags.FailSecondChunk = false;
recorded.Clear();
var svc3 = Build("split", chunks: 1);
await svc3.InterpretTextAsync(text, "probe.pdf", "ro");
Check("ExtractorChunks=1 disables chunking",
    recorded.Count(c => c.Stage == "A") == 1 && recorded.First(c => c.Stage == "A").Fragment == 0);

// ================= 4. Monolithic mode untouched =============================
recorded.Clear();
var svc4 = Build("monolithic");
var (res4, _, _, _) = await svc4.InterpretTextAsync(text, "probe.pdf", "ro");
Check("monolithic: exactly ONE call, no stage prompts, no chunks",
    recorded.Count == 1 && recorded[0].Stage == "MONO");
Check("monolithic: no thinking key sent, primary model",
    recorded[0].Model == "gemini-2.5-flash" && recorded[0].ThinkingBudget == null && recorded[0].ThinkingLevel == null);
Check("monolithic: complete result with its own explanations",
    res4.KeyResults?.Count == 40 && res4.Summary == "Sumar monolitic"
    && res4.KeyResults!.All(k => (k.Explanation ?? "").StartsWith("Explicatie monolitica")));
Check("monolithic: no split telemetry / model label",
    svc4.LastStageTimings.Count == 0 && svc4.LastModelsUsed == "");

// ================= 5. Stage C dies -> monolithic fallback ===================
flags.FailStageC = true;
recorded.Clear();
var svc5 = Build("split");
var (res5, _, _, _) = await svc5.InterpretTextAsync(text, "probe.pdf", "ro");
Check("fallback: monolithic call took over when C failed everywhere",
    recorded.Count(c => c.Stage == "MONO") == 1);
Check("fallback: user still gets a complete report",
    res5.KeyResults?.Count == 40 && res5.Summary == "Sumar monolitic");
Check("fallback: split telemetry cleared", svc5.LastStageTimings.Count == 0);
Check("fallback: the REASON is visible in the Admin Model column",
    svc5.LastModelsUsed.StartsWith("mono <- split fail:") && svc5.LastModelsUsed.Length <= 40,
    svc5.LastModelsUsed);

listener.Stop();

// ================= 6. Deterministic out-of-range completer ==================
var rr = new InterpretationResult
{
    IsMedicalAnalysis = true,
    KeyResults = new List<KeyResult>(),
    AbnormalFindings = new List<AbnormalFinding>()
};
for (int i = 0; i < 20; i++)
    rr.KeyResults.Add(new KeyResult
    {
        Parameter = $"Param {i}", Value = i < 12 ? "150" : "50", Unit = "mg/dL",
        ReferenceRange = "0-100", Status = i < 12 ? "high" : "normal",
        Explanation = $"Explicatia parametrului {i}"
    });
for (int i = 0; i < 8; i++)
    rr.AbnormalFindings.Add(new AbnormalFinding { Parameter = $"Param {i}", Explanation = "de la model", Severity = "mild" });

Check("completer: 8 of 12 becomes 12 of 12",
    AbnormalFindingsCompleter.Complete(rr) == 4 && rr.AbnormalFindings!.Count == 12,
    "count=" + rr.AbnormalFindings!.Count);
Check("completer: no normal analyte leaked in",
    rr.AbnormalFindings.All(f => int.Parse(f.Parameter.Split(' ')[1]) < 12));
Check("completer: model entries untouched", rr.AbnormalFindings.Count(f => f.Explanation == "de la model") == 8);
Check("completer: idempotent", AbnormalFindingsCompleter.Complete(rr) == 0);

var sev = new InterpretationResult
{
    IsMedicalAnalysis = true,
    KeyResults = new List<KeyResult>
    {
        new() { Parameter = "Usor", Value = "110", ReferenceRange = "0-100", Status = "high" },
        new() { Parameter = "Moderat", Value = "130", ReferenceRange = "0-100", Status = "high" },
        new() { Parameter = "Sever", Value = "300", ReferenceRange = "0-100", Status = "high" },
        new() { Parameter = "Scazut", Value = "10", ReferenceRange = "50-100", Status = "low" },
        new() { Parameter = "Borderline", Value = "99", ReferenceRange = "0-100", Status = "borderline" },
        new() { Parameter = "Text", Value = "negativ", ReferenceRange = "negativ", Status = "high" }
    }
};
AbnormalFindingsCompleter.Complete(sev);
string Sev(string p) => sev.AbnormalFindings!.First(f => f.Parameter == p).Severity;
Check("completer: severity mild / moderate / severe computed correctly",
    Sev("Usor") == "mild" && Sev("Moderat") == "moderate" && Sev("Sever") == "severe" && Sev("Scazut") == "severe",
    $"{Sev("Usor")}/{Sev("Moderat")}/{Sev("Sever")}/{Sev("Scazut")}");
Check("completer: borderline and unparseable stay mild",
    Sev("Borderline") == "mild" && Sev("Text") == "mild");
Check("completer: all 6 out-of-range analytes listed", sev.AbnormalFindings!.Count == 6);


// ================= 7. Raw analyte line rebuilt locally ======================
var reportText = string.Join("\n", new[]
{
    "LABORATOR SYNEVO — Buletin de analize",
    "Pacient: Ion Popescu   Data recoltarii: 17/08/2025",
    "",
    "PROFIL LIPIDIC",
    "Colesterol total          250      mg/dL     0 - 200      H   metoda enzimatica",
    "HDL colesterol             38      mg/dL     > 40         L",
    "Trigliceride              210      mg/dL     < 150        H",
    "Nota: colesterolul total se interpreteaza in context clinic.",
    "",
    "HEMATOLOGIE",
    "Hemoglobina              14,2      g/dL      13,0 - 17,0",
    "LLIS Amilaza serica        54      U/L       28 - 100",
    "Feritina                   12      ng/mL     20 - 250     L"
});

var rl = new InterpretationResult
{
    IsMedicalAnalysis = true,
    KeyResults = new List<KeyResult>
    {
        new() { Parameter = "Colesterol total", Value = "250", Unit = "mg/dL", ReferenceRange = "0 - 200", Status = "high" },
        new() { Parameter = "HDL colesterol", Value = "38", Unit = "mg/dL", ReferenceRange = "> 40", Status = "low" },
        new() { Parameter = "Trigliceride", Value = "210", Unit = "mg/dL", ReferenceRange = "< 150", Status = "high" },
        new() { Parameter = "Hemoglobina", Value = "14,2", Unit = "g/dL", ReferenceRange = "13,0 - 17,0", Status = "normal" },
        // name already cleaned of the LLIS marker -> must still find its line
        new() { Parameter = "Amilaza serica", Value = "54", Unit = "U/L", ReferenceRange = "28 - 100", Status = "normal" },
        // model kept its own raw line -> must NOT be overwritten
        new() { Parameter = "Feritina", Value = "12", Unit = "ng/mL", ReferenceRange = "20 - 250", Status = "low",
                AnalyteLineRaw = "linia scrisa de model" },
        // not present in the text -> stays empty, no crash
        new() { Parameter = "Vitamina D", Value = "18", Unit = "ng/mL", ReferenceRange = "30 - 100", Status = "low" }
    }
};

int rebuilt = RawLineReconstructor.Fill(rl, reportText);
Check("raw line: 5 lines rebuilt locally (0 LLM tokens)", rebuilt == 5, "rebuilt=" + rebuilt);
Check("raw line: exact line found for 'Colesterol total' (not the comment mentioning it)",
    rl.KeyResults![0].AnalyteLineRaw!.StartsWith("Colesterol total") && rl.KeyResults[0].AnalyteLineRaw!.Contains("250"),
    rl.KeyResults[0].AnalyteLineRaw ?? "null");
Check("raw line: HDL and Trigliceride matched their own rows",
    rl.KeyResults[1].AnalyteLineRaw!.Contains("38") && rl.KeyResults[2].AnalyteLineRaw!.Contains("210"));
Check("raw line: comma decimals match (Hemoglobina 14,2)",
    rl.KeyResults[3].AnalyteLineRaw!.Contains("14,2"), rl.KeyResults[3].AnalyteLineRaw ?? "null");
Check("raw line: analyte whose name lost the LLIS marker is still found",
    (rl.KeyResults[4].AnalyteLineRaw ?? "").Contains("Amilaza serica"), rl.KeyResults[4].AnalyteLineRaw ?? "null");
Check("raw line: a line already provided by the model is NEVER overwritten",
    rl.KeyResults[5].AnalyteLineRaw == "linia scrisa de model");
Check("raw line: an analyte missing from the text stays empty (no crash, no wrong line)",
    string.IsNullOrEmpty(rl.KeyResults[6].AnalyteLineRaw), rl.KeyResults[6].AnalyteLineRaw ?? "null");
Check("raw line: idempotent (second pass rebuilds nothing)", RawLineReconstructor.Fill(rl, reportText) == 0);
Check("raw line: no text / no analytes is a safe no-op",
    RawLineReconstructor.Fill(rl, null) == 0 && RawLineReconstructor.Fill(null, reportText) == 0
    && RawLineReconstructor.Fill(new InterpretationResult(), reportText) == 0);

// The stage-A prompt must ask to skip the raw line ONLY in TEXT mode.
var aPrompt = St("A")[0].UserText;
Check("raw line: stage A prompt asks to SKIP analyte_line_raw in TEXT mode",
    aPrompt.Contains("SKIP \"analyte_line_raw\"") || aPrompt.Contains("SKIP \u0022analyte_line_raw"),
    aPrompt.Contains("SKIP") ? "found" : "missing");

Console.WriteLine(fails == 0 ? "\nALL CHECKS PASSED" : $"\n{fails} CHECK(S) FAILED");
return fails == 0 ? 0 : 1;

record Call(string Stage, string Model, string? ThinkingLevel, string? ThinkingBudget,
            DateTime Started, DateTime Finished, string UserText, string SystemPrompt, int Fragment);

sealed class Flags
{
    public bool FailSecondChunk;
    public bool FailStageC;
    public bool SweepFinds = true;
}

sealed class ProbeHttpClientFactory : IHttpClientFactory
{
    public HttpClient CreateClient(string name) => new HttpClient();
}
